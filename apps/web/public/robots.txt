# SkillSplore

User-agent: *

# Signed-in and administrative areas. No value in an index, and several would
# 401 or redirect anyway.
Disallow: /account
Disallow: /dashboard
Disallow: /messages
Disallow: /notifications
Disallow: /engagements
Disallow: /admin
# Individual requests and the "my requests" list are private. /requests/new is
# a public landing page and is in sitemap.xml, so it is allowed back in --
# blanket-blocking /requests contradicted the sitemap and had Google fetch a
# URL it was told not to crawl. The longer, more specific rule wins.
Disallow: /requests
Allow: /requests/new
Disallow: /tutor/feed
Disallow: /tutor/responses
Disallow: /verify-email
Disallow: /reset-password
Disallow: /forgot-password

# Everything else -- the homepage, search, category browse, public profiles and
# the policy pages -- is crawlable.
Allow: /

Sitemap: https://skillsplore.org/sitemap.xml
